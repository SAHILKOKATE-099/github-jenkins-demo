pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build') {
            steps {
                bat 'docker build -t github-jenkins-demo .'
            }
        }

        stage('Deploy') {
            steps {
                bat 'docker rm -f github-jenkins-demo || exit /b 0'
                bat 'docker run -d --name github-jenkins-demo -p 5000:5000 github-jenkins-demo'
            }
        }
    }
}
